        function drawGrid() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            const padding = 50;
            const width = canvas.width - 2 * padding;
            const height = canvas.height - 2 * padding;

            ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
            ctx.lineWidth = 1;

            // Hours (0 - 14)
            for (let i = 0; i <= 14; i++) {
                const x = padding + (i * (width / 14));
                ctx.beginPath();
                ctx.moveTo(x, padding);
                ctx.lineTo(x, canvas.height - padding);
                ctx.stroke();

                ctx.fillStyle = '#94a3b8';
                ctx.font = '10px Segoe UI';
                ctx.fillText(i + 'h', x - 6, canvas.height - padding + 20);
            }

            // Dilation (0 - 10 cm)
            for (let i = 0; i <= 10; i++) {
                const y = canvas.height - padding - (i * (height / 10));
                ctx.beginPath();
                ctx.moveTo(padding, y);
                ctx.lineTo(canvas.width - padding, y);
                ctx.stroke();

                ctx.fillStyle = '#94a3b8';
                ctx.font = '10px Segoe UI';
                ctx.fillText(i + ' cm', padding - 35, y + 4);
            }

            // Alert Line
            ctx.strokeStyle = '#ff4b5c';
            ctx.lineWidth = 2;
            ctx.setLineDash([5, 5]);
            ctx.beginPath();
            const xAlertStart = padding + (0 * (width / 14));
            const yAlertStart = canvas.height - padding - (4 * (height / 10));
            const xAlertEnd = padding + (6 * (width / 14));
            const yAlertEnd = canvas.height - padding - (10 * (height / 10));
            ctx.moveTo(xAlertStart, yAlertStart);
            ctx.lineTo(xAlertEnd, yAlertEnd);
            ctx.stroke();

            // Action Line
            ctx.strokeStyle = '#f1c40f';
            ctx.beginPath();
            const xActionStart = padding + (4 * (width / 14));
            const xActionEnd = padding + (10 * (width / 14));
            ctx.moveTo(xActionStart, yAlertStart);
            ctx.lineTo(xActionEnd, yAlertEnd);
            ctx.stroke();
            ctx.setLineDash([]);

            // Points Drawing
            if (dataPoints.length > 0) {
                // Dilation Line (Cyan)
                ctx.strokeStyle = '#00f2fe';
                ctx.lineWidth = 3;
                ctx.beginPath();
                dataPoints.forEach((pt, index) => {
                    const x = padding + (pt.time * (width / 14));
                    const y = canvas.height - padding - (pt.dilation * (height / 10));
                    if (index === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                });
                ctx.stroke();

                // Dilation Points (X)
                dataPoints.forEach(pt => {
                    const x = padding + (pt.time * (width / 14));
                    const y = canvas.height - padding - (pt.dilation * (height / 10));
                    ctx.fillStyle = '#00f2fe';
                    ctx.font = 'bold 14px Segoe UI';
                    ctx.fillText('X', x - 5, y + 5);
                });

                // Station Line (Blue Circle)
                ctx.strokeStyle = '#4facfe';
                ctx.lineWidth = 2;
                ctx.beginPath();
                dataPoints.forEach((pt, index) => {
                    const mappedStation = pt.station + 3; // Mapping -3..+3 to 0..6 grid units
                    const x = padding + (pt.time * (width / 14));
                    const y = canvas.height - padding - (mappedStation * (height / 10));
                    if (index === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                });
                ctx.stroke();

                dataPoints.forEach(pt => {
                    const mappedStation = pt.station + 3;
                    const x = padding + (pt.time * (width / 14));
                    const y = canvas.height - padding - (mappedStation * (height / 10));
                    ctx.fillStyle = '#4facfe';
                    ctx.beginPath();
                    ctx.arc(x, y, 5, 0, Math.PI * 2);
                    ctx.fill();
                });
            }
        }

        function addDataPoint() {
            const time = parseFloat(document.getElementById('timeInput').value);
            const dilation = parseFloat(document.getElementById('dilationInput').value);
            const station = parseInt(document.getElementById('stationInput').value);

            if (isNaN(time) || isNaN(dilation)) return;

            dataPoints.push({ time, dilation, station });
            dataPoints.sort((a, b) => a.time - b.time);

            drawGrid();
        }

        function resetData() {
            dataPoints = [];
            document.getElementById('timeInput').value = 0;
            document.getElementById('dilationInput').value = 4;
            drawGrid();
        }

        // Set default date to today
        document.getElementById('examDate').valueAsDate = new Date();
